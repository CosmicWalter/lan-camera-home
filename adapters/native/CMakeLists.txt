cmake_minimum_required(VERSION 3.16)
project(jiajing_native_adapters LANGUAGES CXX)

set(HIKVISION_SDK_DIR "" CACHE PATH "Extracted Hikvision Device Network SDK")
set(DAHUA_SDK_DIR "" CACHE PATH "Extracted Dahua NetSDK")
set(UNIVIEW_SDK_DIR "" CACHE PATH "Extracted Uniview NetDEVSDK")

find_package(Threads REQUIRED)

function(native_adapter vendor header library root)
    find_path(${vendor}_INCLUDE ${header} HINTS "${root}" PATH_SUFFIXES include Include inc Include/Common NO_DEFAULT_PATH)
    find_library(${vendor}_LIBRARY NAMES ${library} HINTS "${root}" PATH_SUFFIXES lib Lib lib64 Lib/Win64 NO_DEFAULT_PATH)
    if(NOT ${vendor}_INCLUDE OR NOT ${vendor}_LIBRARY)
        message(FATAL_ERROR "Could not find ${header} and ${library} under the specified SDK directory")
    endif()
    add_executable(jiajing-${vendor} ${vendor}.cpp)
    target_compile_features(jiajing-${vendor} PRIVATE cxx_std_17)
    target_include_directories(jiajing-${vendor} SYSTEM PRIVATE "${${vendor}_INCLUDE}")
    target_link_libraries(jiajing-${vendor} PRIVATE "${${vendor}_LIBRARY}" Threads::Threads)
endfunction()

if(HIKVISION_SDK_DIR)
    native_adapter(hikvision HCNetSDK.h "hcnetsdk;HCNetSDK" "${HIKVISION_SDK_DIR}")
endif()
if(DAHUA_SDK_DIR)
    native_adapter(dahua dhnetsdk.h "dhnetsdk" "${DAHUA_SDK_DIR}")
endif()
if(UNIVIEW_SDK_DIR)
    native_adapter(uniview NetDEVSDK.h "NetDEVSDK;netdevsdk" "${UNIVIEW_SDK_DIR}")
endif()
if(NOT (HIKVISION_SDK_DIR OR DAHUA_SDK_DIR OR UNIVIEW_SDK_DIR))
    message(FATAL_ERROR "Specify at least one extracted SDK: HIKVISION_SDK_DIR, DAHUA_SDK_DIR, or UNIVIEW_SDK_DIR")
endif()
